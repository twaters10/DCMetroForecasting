# Repo-level development dependencies: tests, the local ETL, and the notebooks.
#
# Distinct from infra/lambda_collector/requirements-dev.txt, which is scoped to
# the collector and inherits that component's runtime pins. Nothing here is ever
# bundled into the Lambda zip.
#
# This file is also what CI installs (.github/workflows/etl-daily.yml), so the
# data-processing stack is pinned. That job runs unattended every night: an
# unpinned major bump would break it one morning with no change on our side, and
# the failure would look like a data problem rather than a dependency one.

pytest

# --------------------------------------------------------------------------
# The ETL's data stack — pinned
# --------------------------------------------------------------------------
# Stage B. Needs a JVM on the host; `src/etl/spark.py` fails with install
# instructions rather than a stack trace if none is found.
pyspark==4.2.0

# Declared explicitly rather than leaned on as transitive dependencies. `pyspark`
# treats both as optional, but src/etl imports them directly — `decode.py` builds
# Arrow tables and `archive.py`/`schedule.py` use pandas frames — so a clean
# install without these fails at import time.
#
# pandas 3.x makes PySpark emit a "does not yet fully support pandas >= 3.0"
# warning. It is pinned here at the version everything was verified against; the
# ETL avoids the Spark<->pandas bridge entirely (schedule frames go through
# Parquet) precisely so that warning stays cosmetic.
pandas==3.0.5
pyarrow==25.0.1

# Same reasoning, one layer up: the feature modules (`baselines.py`,
# `historical.py`, `safe.py`) import numpy directly rather than reaching it
# through pandas, so it is declared rather than inherited.
numpy==2.5.1

# --------------------------------------------------------------------------
# Feed parsing — matched to the collector
# --------------------------------------------------------------------------
# Pinned to the collector's runtime versions so a snapshot that parses in Lambda
# parses identically here.
gtfs-realtime-bindings==2.2.0
protobuf==7.35.1
requests==2.34.2

python-dotenv
boto3

# --------------------------------------------------------------------------
# Modelling — stage 3
# --------------------------------------------------------------------------
# LightGBM is not an interchangeable choice here. `features/config.py` commits to
# native categorical handling over target encoding (leakage risk for modest lift on
# 785 segments), and native categorical splits are a LightGBM feature.
lightgbm==4.7.0

# Metrics only. Nothing here fits a scikit-learn estimator.
scikit-learn==1.9.0

# --------------------------------------------------------------------------
# Tooling and notebooks
# --------------------------------------------------------------------------
# Writes docs/data_dictionary.xlsx.
openpyxl==3.1.5

# The local UI over the deployed endpoint. Local rather than hosted because a SageMaker
# endpoint needs SigV4 and a static page cannot sign a request; boto3 does it from here,
# which keeps "the only managed compute is the collector Lambda and the endpoint" true.
streamlit==1.62.0

# Run comparison, browsed locally for the same reason Streamlit is: SageMaker offers a
# managed MLflow tracking server, but it bills by the hour to host a UI one person opens
# after a retrain. The tracking store is a derived view — `models/mlflow_sync` rebuilds
# it from the run manifests — so hosting it would buy durability that is not needed and
# would put a third thing in "the only managed compute is the collector Lambda and the
# endpoint". Artifacts still go to S3.
mlflow==3.15.1

black
ruff

# Needed by notebooks/, not by the ETL. Kept in one file rather than split so

jupyter
